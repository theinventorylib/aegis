Aegis provides sophisticated rate limiting to protect against brute force attacks, credential stuffing, and API abuse. Choose between distributed Redis-based limiting for multi-instance deployments or in-memory limiting for single servers.

## Rate Limiting Strategies

### Per-IP Rate Limiting

Limit requests by client IP address to prevent attacks from specific sources:

```go
import "github.com/theinventorylib/aegis/core"

rateLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 100,            // 100 requests max
    WindowDuration:    time.Minute,    // per 1 minute
    KeyPrefix:         "ratelimit:",
    ByIP:              true,           // Enable IP-based limiting
    ByUser:            false,
}, redisClient, auditLogger)

// Use as middleware
router.Use(func(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        allowed, retryAfter, err := rateLimiter.AllowRequest(r.Context(), "")
        if !allowed {
            w.Header().Set("Retry-After", retryAfter.String())
            http.Error(w, "Too many requests", http.StatusTooManyRequests)
            return
        }
        next.ServeHTTP(w, r)
    })
})
```

### Per-User Rate Limiting

Limit authenticated users individually:

```go
rateLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 1000,           // 1000 requests
    WindowDuration:    time.Hour,      // per hour
    ByUser:            true,           // Enable user-based limiting
    ByIP:              false,
}, redisClient, auditLogger)

// Middleware extracts user from context
router.Use(func(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        user, err := core.GetUser(r.Context())
        if err == nil {
            allowed, retryAfter, _ := rateLimiter.AllowRequest(r.Context(), user.GetID())
            if !allowed {
                http.Error(w, "Rate limit exceeded", http.StatusTooManyRequests)
                return
            }
        }
        next.ServeHTTP(w, r)
    })
})
```

### Combined IP + User Limiting

Apply both limits simultaneously for maximum protection:

```go
rateLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 50,
    WindowDuration:    time.Minute,
    ByIP:              true,   // Limit by IP
    ByUser:            true,   // AND limit by user
}, redisClient, auditLogger)

// Both IP and user must be within limits
// If either limit is exceeded, request is denied
```

## Redis-Based Distributed Limiting

For multi-instance deployments, use Redis to share rate limit state:

### Setup

```go
import "github.com/redis/go-redis/v9"

// Create Redis client
redisClient := redis.NewClient(&redis.Options{
    Addr:     "localhost:6379",
    Password: "",
    DB:       0,
})

// Verify connection
if err := redisClient.Ping(ctx).Err(); err != nil {
    log.Fatal("Redis connection failed:", err)
}

// Create distributed rate limiter
rateLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 100,
    WindowDuration:    time.Minute,
    KeyPrefix:         "ratelimit:",
    ByIP:              true,
}, redisClient, auditLogger)  // Pass Redis client
```

### How It Works

```go
// Redis stores counters with TTL
// Key: ratelimit:ip:192.168.1.100
// Value: request count
// TTL: window duration

func (rl *RateLimiter) AllowRequest(ctx context.Context, key string) (bool, time.Duration, error) {
    redisKey := rl.config.KeyPrefix + key
    
    // Increment counter
    count, err := rl.redisClient.Incr(ctx, redisKey).Result()
    if err != nil {
        return false, 0, err
    }
    
    // Set expiration on first request
    if count == 1 {
        rl.redisClient.Expire(ctx, redisKey, rl.config.WindowDuration)
    }
    
    // Check limit
    if count > int64(rl.config.RequestsPerWindow) {
        ttl, _ := rl.redisClient.TTL(ctx, redisKey).Result()
        return false, ttl, nil
    }
    
    return true, 0, nil
}
```

### Benefits

- **Shared State**: All application instances share the same counters
- **Accurate Limits**: No per-instance splitting required
- **Fast**: Sub-millisecond Redis operations
- **Scalable**: Handles millions of requests per second

## In-Memory Fallback

When Redis is unavailable or not configured, Aegis uses in-memory rate limiting:

### Automatic Fallback

```go
// No Redis client provided → in-memory mode
rateLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 100,
    WindowDuration:    time.Minute,
    ByIP:              true,
}, nil, auditLogger)  // nil = in-memory fallback
```

### In-Memory Implementation

```go
type RateLimiter struct {
    memoryStore     map[string]*rateLimitEntry
    memoryStoreMu   sync.RWMutex
    cleanupTicker   *time.Ticker
}

type rateLimitEntry struct {
    count     int
    expiresAt time.Time
}

func (rl *RateLimiter) allowRequestMemory(key string) (bool, time.Duration) {
    rl.memoryStoreMu.Lock()
    defer rl.memoryStoreMu.Unlock()
    
    entry, exists := rl.memoryStore[key]
    now := time.Now()
    
    if !exists || now.After(entry.expiresAt) {
        // Create new window
        rl.memoryStore[key] = &rateLimitEntry{
            count:     1,
            expiresAt: now.Add(rl.config.WindowDuration),
        }
        return true, 0
    }
    
    // Increment counter
    entry.count++
    
    if entry.count > rl.config.RequestsPerWindow {
        return false, entry.expiresAt.Sub(now)
    }
    
    return true, 0
}
```

### Automatic Cleanup

```go
// Background goroutine cleans expired entries
func (rl *RateLimiter) startCleanup() {
    rl.cleanupTicker = time.NewTicker(time.Minute)
    rl.cleanupStopChan = make(chan struct{})
    
    go func() {
        for {
            select {
            case <-rl.cleanupTicker.C:
                rl.cleanupExpired()
            case <-rl.cleanupStopChan:
                return
            }
        }
    }()
}

func (rl *RateLimiter) cleanupExpired() {
    rl.memoryStoreMu.Lock()
    defer rl.memoryStoreMu.Unlock()
    
    now := time.Now()
    for key, entry := range rl.memoryStore {
        if now.After(entry.expiresAt) {
            delete(rl.memoryStore, key)
        }
    }
}
```

::callout{icon="i-heroicons-information-circle" color="blue"}
**Single Instance Only**: In-memory limiting only works for single-server deployments. For load-balanced or clustered setups, use Redis.
::

## Login Attempt Tracking

Prevent brute force attacks on user accounts with login attempt tracking:

### Configuration

```go
loginAttemptTracker := core.NewLoginAttemptTracker(
    redisClient,        // Use Redis for distributed tracking
    5,                  // Max 5 failed attempts
    15*time.Minute,     // Lockout for 15 minutes
    time.Hour,          // Track attempts for 1 hour
)

authService := core.NewAuthService(/* ... */)
authService.SetLoginAttemptTracker(loginAttemptTracker)
```

### How It Works

```go
// On login attempt
func (h *EmailPasswordHandlers) Login(ctx context.Context, email, password string) (*LoginResult, error) {
    // 1. Check if account is locked
    locked, remaining, err := h.authService.loginAttemptTracker.IsLockedOut(ctx, email)
    if locked {
        return nil, fmt.Errorf("account locked, try again in %s", remaining)
    }
    
    // 2. Verify credentials
    user, err := h.authService.User.GetUserByEmail(ctx, email)
    if err != nil {
        // 3. Record failed attempt
        attempts, lockout, _ := h.authService.loginAttemptTracker.RecordFailedAttempt(ctx, email)
        if lockout {
            return nil, errors.New("too many failed attempts, account locked")
        }
        return nil, ErrInvalidCredentials
    }
    
    valid, err := h.authService.Account.VerifyPassword(ctx, user.GetID(), password)
    if !valid {
        // 4. Record failed attempt
        h.authService.loginAttemptTracker.RecordFailedAttempt(ctx, email)
        return nil, ErrInvalidCredentials
    }
    
    // 5. Clear attempts on success
    h.authService.loginAttemptTracker.ClearAttempts(ctx, email)
    
    // Create session...
    return result, nil
}
```

### Programmatic Usage

```go
// Check if user is locked out
locked, remaining, err := tracker.IsLockedOut(ctx, "user@example.com")
if locked {
    fmt.Printf("Account locked for %s\n", remaining)
}

// Record failed login
attempts, lockout, err := tracker.RecordFailedAttempt(ctx, "user@example.com")
if lockout {
    fmt.Printf("Account locked after %d attempts\n", attempts)
}

// Clear attempts (successful login)
err = tracker.ClearAttempts(ctx, "user@example.com")

// Get current attempt count
attempts, err := tracker.GetAttemptCount(ctx, "user@example.com")
```

## Account Lockout Protection

### Configurable Thresholds

```go
// Strict: Lock after 3 attempts for 30 minutes
strictTracker := core.NewLoginAttemptTracker(
    redisClient,
    3,              // 3 attempts
    30*time.Minute, // 30 min lockout
    time.Hour,
)

// Lenient: Lock after 10 attempts for 5 minutes
lenientTracker := core.NewLoginAttemptTracker(
    redisClient,
    10,             // 10 attempts
    5*time.Minute,  // 5 min lockout
    24*time.Hour,   // Track for 24 hours
)

// Progressive: Increase lockout duration
type ProgressiveLockout struct {
    tracker     *core.LoginAttemptTracker
    lockoutDurations []time.Duration
}

func (p *ProgressiveLockout) GetLockoutDuration(attempts int) time.Duration {
    if attempts <= 3 {
        return 5 * time.Minute
    } else if attempts <= 6 {
        return 15 * time.Minute
    } else if attempts <= 10 {
        return time.Hour
    }
    return 24 * time.Hour  // Long lockout after 10+ attempts
}
```

### Admin Override

```go
// Unlock user account manually (admin action)
func unlockAccount(ctx context.Context, email string) error {
    return loginAttemptTracker.ClearAttempts(ctx, email)
}

// Get lockout status for admin dashboard
func getAccountStatus(ctx context.Context, email string) map[string]any {
    locked, remaining, _ := loginAttemptTracker.IsLockedOut(ctx, email)
    attempts, _ := loginAttemptTracker.GetAttemptCount(ctx, email)
    
    return map[string]any{
        "locked":    locked,
        "remaining": remaining,
        "attempts":  attempts,
    }
}
```

## Route-Specific Rate Limits

Apply different limits to different endpoints:

```go
// Strict limit for login endpoint
loginLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 5,
    WindowDuration:    time.Minute,
    ByIP:              true,
}, redisClient, auditLogger)

// Lenient limit for general API
apiLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 100,
    WindowDuration:    time.Minute,
    ByIP:              true,
}, redisClient, auditLogger)

// Apply to routes
router.Post("/auth/default/login", applyRateLimit(loginLimiter), handleLogin)
router.Get("/api/data", applyRateLimit(apiLimiter), handleAPIRequest)

func applyRateLimit(limiter *core.RateLimiter) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            allowed, retryAfter, _ := limiter.AllowRequest(r.Context(), "")
            if !allowed {
                w.Header().Set("Retry-After", retryAfter.String())
                http.Error(w, "Rate limit exceeded", http.StatusTooManyRequests)
                return
            }
            next.ServeHTTP(w, r)
        })
    }
}
```

## Excluding Paths

Exempt certain routes from rate limiting:

```go
rateLimiter := core.NewRateLimiter(&core.RateLimitConfig{
    RequestsPerWindow: 100,
    WindowDuration:    time.Minute,
    ByIP:              true,
    ExcludePaths: []string{
        "/health",          // Health check endpoint
        "/metrics",         // Prometheus metrics
        "/auth/default/logout",  // Allow unlimited logouts
    },
}, redisClient, auditLogger)

// Middleware checks exclude list
func (rl *RateLimiter) shouldExclude(path string) bool {
    for _, excludePath := range rl.config.ExcludePaths {
        if strings.HasPrefix(path, excludePath) {
            return true
        }
    }
    return false
}
```

## Advanced Examples

### Tiered Rate Limiting

Different limits for different user tiers:

```go
type TieredRateLimiter struct {
    freeLimiter    *core.RateLimiter
    premiumLimiter *core.RateLimiter
    userService    *core.UserService
}

func (t *TieredRateLimiter) AllowRequest(ctx context.Context, user auth.User) (bool, time.Duration, error) {
    // Check user tier
    isPremium := checkPremiumStatus(user)
    
    if isPremium {
        return t.premiumLimiter.AllowRequest(ctx, user.GetID())
    }
    return t.freeLimiter.AllowRequest(ctx, user.GetID())
}

// Usage
tieredLimiter := &TieredRateLimiter{
    freeLimiter: core.NewRateLimiter(&core.RateLimitConfig{
        RequestsPerWindow: 100,
        WindowDuration:    time.Hour,
    }, redis, nil),
    premiumLimiter: core.NewRateLimiter(&core.RateLimitConfig{
        RequestsPerWindow: 10000,
        WindowDuration:    time.Hour,
    }, redis, nil),
}
```

### Dynamic Rate Limiting

Adjust limits based on system load:

```go
type DynamicRateLimiter struct {
    baseLimiter *core.RateLimiter
    loadMonitor *LoadMonitor
}

func (d *DynamicRateLimiter) GetCurrentLimit() int {
    load := d.loadMonitor.GetCurrentLoad()
    
    if load > 0.9 {
        return 10  // Severe throttling
    } else if load > 0.7 {
        return 50  // Moderate throttling
    }
    return 100  // Normal rate
}
```

### Geographic Rate Limiting

Different limits by country or region:

```go
type GeoRateLimiter struct {
    limiters map[string]*core.RateLimiter
    geoIP    *GeoIPService
}

func (g *GeoRateLimiter) AllowRequest(ctx context.Context, ip string) (bool, time.Duration, error) {
    country := g.geoIP.GetCountry(ip)
    
    limiter, ok := g.limiters[country]
    if !ok {
        limiter = g.limiters["default"]
    }
    
    return limiter.AllowRequest(ctx, ip)
}
```

## Monitoring and Metrics

### Audit Logging

```go
type RateLimitAuditLogger struct {
    logger *log.Logger
}

func (l *RateLimitAuditLogger) LogRateLimitExceeded(ctx context.Context, key string, limit int) error {
    meta := core.GetRequestMeta(ctx)
    l.logger.Printf(
        "[RATE_LIMIT] Exceeded - Key: %s, Limit: %d, IP: %s, Path: %s",
        key, limit, meta.IPAddress, meta.Path,
    )
    return nil
}
```

### Prometheus Metrics

```go
import "github.com/prometheus/client_golang/prometheus"

var (
    rateLimitExceeded = prometheus.NewCounterVec(
        prometheus.CounterOpts{
            Name: "aegis_rate_limit_exceeded_total",
            Help: "Total number of rate limit violations",
        },
        []string{"key_type"},  // "ip", "user"
    )
    
    rateLimitRequests = prometheus.NewCounterVec(
        prometheus.CounterOpts{
            Name: "aegis_rate_limit_requests_total",
            Help: "Total number of rate-checked requests",
        },
        []string{"key_type", "allowed"},
    )
)

func recordRateLimitMetrics(keyType string, allowed bool) {
    rateLimitRequests.WithLabelValues(keyType, fmt.Sprint(allowed)).Inc()
    if !allowed {
        rateLimitExceeded.WithLabelValues(keyType).Inc()
    }
}
```
