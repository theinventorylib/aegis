## Testing Your Auth Implementation

Proper testing ensures your authentication system works correctly and securely.

## Unit Testing

### Testing Handlers

```go
func TestSignupHandler(t *testing.T) {
    // Setup test database
    db := setupTestDB(t)
    defer db.Close()
    
    mux := chi.NewRouter()
    r := router.NewChiRouter(mux)
    
    cfg := config.Default().
        WithDB(db).
        WithRouter(r).
        WithSecret([]byte("test-secret-32-characters-long!!"))

    a, err := aegis.New(context.Background(), cfg)
    if err != nil {
        t.Fatal(err)
    }
    a.MountRoutes("/auth")
    
    // Create test request
    body := strings.NewReader(`{"name":"Test","email":"test@example.com","password":"password123"}`)
    req := httptest.NewRequest("POST", "/auth/default/signup", body)
    req.Header.Set("Content-Type", "application/json")
    
    // Create response recorder
    w := httptest.NewRecorder()
    
    // Serve via the chi router
    mux.ServeHTTP(w, req)
    
    // Assert response
    if w.Code != http.StatusOK {
        t.Errorf("Expected 200, got %d", w.Code)
    }
}
```

## Integration Testing

### Testing Full Auth Flow

```go
func TestAuthFlow(t *testing.T) {
    // Start test server
    ts := httptest.NewServer(setupRouter())
    defer ts.Close()
    
    client := &http.Client{
        Jar: cookiejar.New(nil),
    }
    
    // 1. Sign up
    resp, err := client.Post(ts.URL+"/auth/default/signup",
        "application/json",
        strings.NewReader(`{"name":"Test","email":"test@example.com","password":"pass123"}`))
    
    if err != nil {
        t.Fatal(err)
    }
    
    if resp.StatusCode != http.StatusOK {
        t.Errorf("Signup failed: %d", resp.StatusCode)
    }
    
    // 2. Sign in
    resp, err = client.Post(ts.URL+"/auth/default/login",
        "application/json",
        strings.NewReader(`{"email":"test@example.com","password":"pass123"}`))
    
    if err != nil {
        t.Fatal(err)
    }
    
    // 3. Access protected route
    resp, err = client.Get(ts.URL+"/api/profile")
    
    if err != nil {
        t.Fatal(err)
    }
    
    if resp.StatusCode != http.StatusOK {
        t.Errorf("Protected route failed: %d", resp.StatusCode)
    }
}
```

### Using Docker for Tests

```go
func setupPostgresDB(t *testing.T) *sql.DB {
    pool, err := dockertest.NewPool("")
    if err != nil {
        t.Fatal(err)
    }
    
    resource, err := pool.Run("postgres", "15", []string{
        "POSTGRES_PASSWORD=password",
        "POSTGRES_DB=test",
    })
    if err != nil {
        t.Fatal(err)
    }
    
    t.Cleanup(func() {
        pool.Purge(resource)
    })
    
    var db *sql.DB
    if err := pool.Retry(func() error {
        var err error
        db, err = sql.Open("postgres",
            fmt.Sprintf("postgres://postgres:password@localhost:%s/test?sslmode=disable",
                resource.GetPort("5432/tcp")))
        if err != nil {
            return err
        }
        return db.Ping()
    }); err != nil {
        t.Fatal(err)
    }
    
    return db
}
```

## Table-Driven Tests

```go
func TestPasswordValidation(t *testing.T) {
    tests := []struct {
        name     string
        password string
        wantErr  bool
    }{
        {"valid password", "SecurePass123!", false},
        {"too short", "short", true},
        {"no numbers", "PasswordOnly", true},
        {"no uppercase", "password123", true},
        {"empty", "", true},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            err := core.ValidatePassword(tt.password)
            if (err != nil) != tt.wantErr {
                t.Errorf("ValidatePassword() error = %v, wantErr %v", err, tt.wantErr)
            }
        })
    }
}
```

## Coverage

Run tests with coverage:

```bash
# Run all tests with coverage
go test -cover ./...

# Generate coverage report
go test -coverprofile=coverage.out ./...
go tool cover -html=coverage.out

# Check coverage percentage
go test -cover ./... | grep coverage
```

## CI/CD Testing

### GitHub Actions Example

```yaml
name: Tests

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    
    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: test
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432
    
    steps:
      - uses: actions/checkout@v3
      
      - name: Set up Go
        uses: actions/setup-go@v4
        with:
          go-version: '1.25'
      
      - name: Run tests
        env:
          DB_URL: postgres://postgres:postgres@localhost:5432/test?sslmode=disable
        run: |
          go test -v -race -coverprofile=coverage.out ./...
          go tool cover -func=coverage.out
```

## Best Practices

1. **Use test databases**: Never test against production databases
2. **Clean up**: Always clean up test data and resources
3. **Parallel tests**: Use `t.Parallel()` for independent tests
4. **Table-driven**: Use table-driven tests for multiple scenarios
5. **Mock external services**: Don't make real API calls in tests
6. **Test errors**: Test both success and failure cases
7. **Measure coverage**: Aim for >80% code coverage
8. **CI integration**: Run tests automatically on every commit
